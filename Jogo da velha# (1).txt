import java.util.Random;
import java.util.Scanner;

/**
 *
 * @author fef
 */
public class JavaApplication1 {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Random gerador = new Random();
        Scanner ler = new Scanner(System.in);
        
        char[][] tabuleiro = {
            {' ', ' ', ' '},
            {' ', ' ', ' '},
            {' ', ' ', ' '}
        };

        char jogador = 'X';
        char sistema = 'O';

        int jogadas = 0;
        boolean venceu = false;
        
        System.out.println("=== JOGO DA VELHA ===");
        mostrarTabuleiro(tabuleiro);

        while (jogadas < 9 && !venceu) {

            System.out.println("\nSua vez! Você é o X.");

            int linha;
            int coluna;

            while (true) {

                System.out.print("Digite a linha (1-3): ");
                linha = ler.nextInt() - 1;

                System.out.print("Digite a coluna (1-3): ");
                coluna = ler.nextInt() - 1;

                if (linha >= 0 && linha < 3 &&
                    coluna >= 0 && coluna < 3 &&
                    tabuleiro[linha][coluna] == ' ') {

                    break;

                } else {

                    System.out.println("Posição inválida ou ocupada!");
                    System.out.println("Escolha outra posição.\n");
                }
            }

            tabuleiro[linha][coluna] = jogador;
            jogadas++;

            System.out.println("\nVocê jogou:");
            mostrarTabuleiro(tabuleiro);

            venceu = verificarVitoria(tabuleiro, jogador);

            if (venceu) {
                break;
            }

            if (jogadas == 9) {
                break;
            }

            System.out.println("\nO sistema está pensando...");

            int linhaSistema;
            int colunaSistema;

            do {

                linhaSistema = gerador.nextInt(3);
                colunaSistema = gerador.nextInt(3);

            } while (tabuleiro[linhaSistema][colunaSistema] != ' ');

            tabuleiro[linhaSistema][colunaSistema] = sistema;
            jogadas++;

            System.out.println("O sistema jogou!");

            mostrarTabuleiro(tabuleiro);
            
            venceu = verificarVitoria(tabuleiro, sistema);
        }

        System.out.println("\n=== FIM DE JOGO ===");

        if (verificarVitoria(tabuleiro, jogador)) {

            System.out.println("PARABÉNS! Você venceu!");

        } else if (verificarVitoria(tabuleiro, sistema)) {

            System.out.println("O sistema venceu!");

        } else {

            System.out.println("EMPATE!");
        }
    }

    public static void mostrarTabuleiro(char[][] tabuleiro) {

        System.out.println();
        System.out.println("    1   2   3");
        System.out.println("  +---+---+---+");

        for (int i = 0; i < 3; i++) {

            System.out.print((i + 1) + " |");

            for (int j = 0; j < 3; j++) {

                System.out.print(" " + tabuleiro[i][j] + " |");
            }

            System.out.println();
            System.out.println("  +---+---+---+");
        }
    }

    public static boolean verificarVitoria(char[][] tabuleiro, char jogador) {

        for (int i = 0; i < 3; i++) {

            if (tabuleiro[i][0] == jogador &&
                tabuleiro[i][1] == jogador &&
                tabuleiro[i][2] == jogador) {

                return true;
            }
        }

        for (int j = 0; j < 3; j++) {

            if (tabuleiro[0][j] == jogador &&
                tabuleiro[1][j] == jogador &&
                tabuleiro[2][j] == jogador) {

                return true;
            }
        }

        if (tabuleiro[0][0] == jogador &&
            tabuleiro[1][1] == jogador &&
            tabuleiro[2][2] == jogador) {

            return true;
        }

        if (tabuleiro[0][2] == jogador &&
            tabuleiro[1][1] == jogador &&
            tabuleiro[2][0] == jogador) {
        }

        return false;
    }
}
